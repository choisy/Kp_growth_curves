A function that plots the data:

plot_data <- function(x, y = OD, ylab = "optical density OD600 nm") {
  x |>
    mutate(across(time, ~ divide_by(.x, 60))) |> 
    slice_sample(prop = 1) |> 
    rename(y = {{ y }}) |> 
    with(plot(time, y, col = col_set_alpha[Medium],
              xlab = "time (hours)", ylab = ylab))
  legend("topleft", legend = col_set_names, pch = 1, bty = "n", col = col_set)
}

6 Model

The logistic growth model:

logistic <- function(t, K, r, t0) {
  K / (1 + exp(-(t - t0) * r))
}

A reparameterization of the gamma density of probability:

dgamma2 <- function(x, mu, k) {
    dgamma(x, k, k / mu, log = TRUE)
}

A reparameterization of the log-normal density of probability:

dlnorm2 <- function(x, mu, k) {
    dlnorm(x, log(mu), k, log = TRUE)
}

The minimum log-likelihood function factory:

make_mLL <- function(x, f) {
  function(K, r, t0, k) {
    - sum(f(x$OD, logistic(x$time, exp(K), exp(r), exp(t0)), exp(k)))  
  }
}

A function that looks for good initial values for the parameters of the logistic growth model:

ini_val <- function(df, k = 1) {
  time <- df$time
  K <- max(df$OD)
  halfK <- K / 2
  evaluate <- function(x) approx(time, df$OD, x)$y
  distance <- function(x) abs(evaluate(x) - halfK)
  t0 <- optimize(distance, range(time))$minimum
  r <- df |> 
    filter(time < t0) |> 
    mutate(across(OD, log)) |> 
    lm(OD ~ time, data = _) |> 
    coef() |> 
    extract(2) |> 
    unname()
  as.list(log(c(K = K, r = r, t0 = t0, k = k)))
}

A tuning of the mle2() function that includes the automatic initial values of the parameters search and the specification of the density of probability we want to use for the minus log-likelihood function:

mle3 <- function(x, f, k = 10, ...) {
  mle2(make_mLL(x, f), ini_val(x, k), "Nelder-Mead", control = list(maxit = 1e4), ...)
}
