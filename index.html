Loading the data:

(od_data <- data_file_full |>
  read_excel() |> 
  group_by(Sample, Medium) |> 
  mutate(replicate = row_number()) |>
  ungroup() |> 
  pivot_longer(- c(Sample, Medium, replicate), names_to = "time", values_to = "OD") |> 
  mutate(across(time, ~ .x |>
                  stringr::str_remove(" *h$") |> 
                  as.numeric() |> 
                  multiply_by(60) |> 
                  round() |> 
                  as.integer())))
